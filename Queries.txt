
CREATE DATABASE library_management;

USE library_management;

CREATE TABLE books
(
    book_id UInt32,
    title String,
    author String,
    category String,
    available_copies UInt32
)
ENGINE = MergeTree
ORDER BY book_id;

CREATE TABLE members
(
    member_id UInt32,
    member_name String,
    department String
)
ENGINE = MergeTree
ORDER BY member_id;

CREATE TABLE borrowings
(
    borrowing_id UInt32,
    member_id UInt32,
    book_id UInt32,
    borrow_date Date
)
ENGINE = MergeTree
ORDER BY borrowing_id;

SELECT
    m.member_name,
    m.department,
    b.title,
    br.borrow_date
FROM members AS m
INNER JOIN borrowings AS br
    ON m.member_id = br.member_id
INNER JOIN book AS b
    ON br.book_id = b.book_Id
ORDER BY br.borrowing_id;

SELECT
    m.member_name,
    m.department,
    br.borrowing_id,
    br.book_id
FROM members AS m
LEFT JOIN borrowings AS br
    ON m.member_id = br.member_id
ORDER BY m.member_id;

SELECT
    b.title,
    br.borrowing_id,
    br.book_id
FROM book AS b
RIGHT JOIN borrowings AS br
    ON b.book_Id = br.book_id
ORDER BY br.borrowing_id;

SELECT
    b.book_Id,
    b.title,
    br.borrowing_id,
    br.borrow_date
FROM book AS b
FULL OUTER JOIN borrowings AS br
    ON b.book_Id = br.book_id
ORDER BY b.book_Id;
